import React, { useState, useEffect, useRef } from 'react';
import { useQuery, useQueryClient } from '@tanstack/react-query';
import api from '../services/api';
import { useOrders, useMenu } from '../hooks/useOrders';
import { useSocket } from '../hooks/useSocket';
import { useAuth } from '../hooks/useAuth';
import DutyRoster from './DutyRoster';
import { 
  LayoutDashboard, 
  UtensilsCrossed, 
  CalendarDays, 
  MessageSquare, 
  History, 
  CreditCard, 
  Settings,
  Search,
  Bell,
  ChevronDown,
  Filter,
  Plus,
  Minus,
  Trash2,
  X,
  CreditCard as CardIcon,
  Wallet,
  Banknote,
  Receipt,
  ArrowRight,
  Star
} from 'lucide-react';
import { motion, AnimatePresence } from 'framer-motion';

const SERVER_URL = 'http://localhost:5000';

const CATEGORY_ICONS: Record<string, string> = {
  'Donuts': '🍩',
  'Burger': '🍔',
  'Ice': '🍦',
  'Potato': '🍟',
  'Pizza': '🍕',
  'Hot dog': '🌭',
  'Chicken': '🍗',
  'Default': '🍴'
};

const WaiterDashboard = () => {
  const { user } = useAuth();
  const [activeView, setActiveView] = useState<'dashboard' | 'menu' | 'roster' | 'messages' | 'history'>('menu');
  const [selectedTableId, setSelectedTableId] = useState<string>('');
  const [cart, setCart] = useState<any[]>([]);
  const [search, setSearch] = useState('');
  
  const queryClient = useQueryClient();
  const { socket } = useSocket();
  const { fetchTables, createOrder, fetchMyOrders, confirmPayment, isCreating } = useOrders();
  const { fetchHierarchicalMenu, fetchCategories } = useMenu();

  // Navigation State for Menu
  const [navPath, setNavPath] = useState<{ type: 'root' | 'menu' | 'category' | 'subcategory', id: string | null, name: string }[]>([{ type: 'root', id: null, name: 'Menu' }]);
  
  // Selection
  const [activeCategory, setActiveCategory] = useState<string | null>(null);

  const { data: tables } = useQuery({ queryKey: ['tables'], queryFn: fetchTables });
  const { data: menus } = useQuery({ queryKey: ['hierarchical-menu'], queryFn: fetchHierarchicalMenu });
  const { data: categories } = useQuery({ queryKey: ['categories'], queryFn: fetchCategories });
  const { data: myOrders } = useQuery({ queryKey: ['my-orders'], queryFn: fetchMyOrders, refetchInterval: 5000 });

  useEffect(() => {
    if (socket) {
      const handleUpdate = () => {
        queryClient.invalidateQueries({ queryKey: ['my-orders'] });
        queryClient.invalidateQueries({ queryKey: ['tables'] });
      };
      socket.on('order_status_update', handleUpdate);
      return () => { socket.off('order_status_update', handleUpdate); };
    }
  }, [socket, queryClient]);

  // Logic helpers
  const currentLevel = navPath[navPath.length - 1];
  
  const getCurrentDisplayData = () => {
    // If we have an active category selected from the top bar
    if (activeCategory) {
       const cat = (categories as any[])?.find(c => c.id === activeCategory);
       return { type: 'items', data: cat?.items || [] };
    }

    if (currentLevel.type === 'root') {
      return { type: 'items', data: (categories as any[])?.flatMap(c => c.items || []) || [] };
    }
    // ... other drill down logic can be added here if needed, 
    // but the Figma design uses a flatter category approach.
    return { type: 'items', data: [] };
  };

  const addToCart = (item: any) => {
    setCart(prev => {
      const existing = prev.find(i => i.id === item.id);
      if (existing) return prev.map(i => i.id === item.id ? { ...i, quantity: i.quantity + 1 } : i);
      return [...prev, { ...item, quantity: 1 }];
    });
  };

  const removeFromCart = (itemId: string) => {
    setCart(prev => prev.filter(i => i.id !== itemId));
  };

  const updateQuantity = (itemId: string, delta: number) => {
    setCart(prev => prev.map(i => 
      i.id === itemId ? { ...i, quantity: Math.max(1, i.quantity + delta) } : i
    ));
  };

  const subTotal = cart.reduce((acc, item) => acc + (item.price * item.quantity), 0);
  const tax = subTotal * 0.05;
  const total = subTotal + tax;

  const display = getCurrentDisplayData();

  return (
    <div className="fixed inset-0 flex bg-[#F9FAFB] text-[#1F2937] antialiased">
      {/* SIDEBAR NAVIGATION */}
      <aside className="w-[240px] bg-white border-r border-[#E5E7EB] flex flex-col pt-8 pb-4 shrink-0">
        <div className="px-8 mb-10 flex items-center gap-3">
          <div className="w-8 h-8 bg-[#FA5D3E] rounded-lg flex items-center justify-center text-white">
            <UtensilsCrossed size={18} />
          </div>
          <span className="font-black text-lg tracking-tight text-[#FA5D3E]">foodislice</span>
        </div>

        <nav className="flex-1 px-4 space-y-1">
          {[
            { id: 'dashboard', icon: LayoutDashboard, label: 'Dashboard' },
            { id: 'menu', icon: UtensilsCrossed, label: 'Food Order' },
            { id: 'messages', icon: MessageSquare, label: 'Message' },
            { id: 'history', icon: History, label: 'Order History' },
            { id: 'roster', icon: CalendarDays, label: 'Duty Roster' },
            { id: 'payments', icon: CreditCard, label: 'Payment details' },
            { id: 'settings', icon: Settings, label: 'Customization' },
          ].map((item: any) => (
            <button
              key={item.id}
              onClick={() => setActiveView(item.id)}
              className={`w-full flex items-center gap-4 px-4 py-3.5 rounded-xl transition-all font-semibold text-sm ${
                activeView === item.id 
                ? 'bg-[#FA5D3E]/10 text-[#FA5D3E]' 
                : 'text-[#9CA3AF] hover:bg-[#F9FAFB] hover:text-[#4B5563]'
              }`}
            >
              <item.icon size={20} />
              {item.label}
            </button>
          ))}
        </nav>

        <div className="mt-auto px-4">
           <div className="bg-[#FA5D3E]/5 rounded-2xl p-4 relative overflow-hidden">
             <div className="relative z-10">
                <p className="text-[11px] font-bold text-[#FA5D3E] uppercase tracking-wider mb-1">New Promo</p>
                <h4 className="text-[13px] font-bold text-slate-800 leading-tight">Order now and get 20% off!</h4>
             </div>
             <div className="w-12 h-12 bg-[#FA5D3E]/20 rounded-full absolute -top-4 -right-4" />
           </div>
        </div>
      </aside>

      {/* MAIN CONTENT AREA */}
      <main className="flex-1 flex flex-col overflow-hidden">
        {/* HEADER */}
        <header className="h-[80px] bg-white border-b border-[#E5E7EB] flex items-center justify-between px-8 shrink-0">
          <div className="w-[400px] relative">
            <Search className="absolute left-4 top-1/2 -translate-y-1/2 text-[#9CA3AF]" size={18} />
            <input 
              type="text" 
              placeholder="Search food" 
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              className="w-full h-[48px] pl-12 pr-4 bg-[#F9FAFB] rounded-xl outline-none text-sm font-medium border-transparent focus:border-[#FA5D3E] transition-all"
            />
          </div>

          <div className="flex items-center gap-6">
            <button className="w-10 h-10 flex items-center justify-center bg-[#F9FAFB] rounded-xl text-[#4B5563] relative">
              <Bell size={20} />
              <span className="w-2 h-2 bg-[#FA5D3E] rounded-full absolute top-2.5 right-3 border-2 border-white" />
            </button>
            <div className="flex items-center gap-3 pl-6 border-l border-[#E5E7EB]">
              <img src="https://ui-avatars.com/api/?name=Waiter+User&background=FA5D3E&color=fff" className="w-10 h-10 rounded-xl" alt="profile" />
              <div className="text-left">
                <p className="text-sm font-bold text-[#1F2937] leading-none mb-1">{user?.name || 'David Brown'}</p>
                <p className="text-[11px] font-medium text-[#9CA3AF] uppercase tracking-wider">{user?.role || 'Waiter'}</p>
              </div>
              <ChevronDown size={16} className="text-[#9CA3AF]" />
            </div>
          </div>
        </header>

        {/* VIEWS */}
        <div className="flex-1 overflow-y-auto p-4 md:p-8 custom-scrollbar">
          {activeView === 'menu' && (
            <div className="space-y-10">
              {/* CATEGORIES BAR */}
              <div className="space-y-4">
                <div className="flex items-center justify-between">
                   <h2 className="text-xl font-black tracking-tight">Explore Categories</h2>
                   <button className="p-2 bg-[#FA5D3E] text-white rounded-lg flex items-center gap-2 text-xs font-bold px-4">
                     Filter <Filter size={14} />
                   </button>
                </div>
                <div className="flex gap-4 overflow-x-auto pb-2 scrollbar-none">
                   <button 
                     onClick={() => setActiveCategory(null)}
                     className={`flex items-center gap-3 px-6 py-3 rounded-xl border transition-all shrink-0 font-bold text-sm ${!activeCategory ? 'bg-[#FA5D3E] border-[#FA5D3E] text-white shadow-lg shadow-[#FA5D3E]/20' : 'bg-white border-[#E5E7EB] text-[#4B5563] hover:border-[#FA5D3E]'}`}
                   >
                     ⭐ All
                   </button>
                   {categories?.map((cat: any) => (
                     <button
                       key={cat.id}
                       onClick={() => setActiveCategory(cat.id)}
                       className={`flex items-center gap-3 px-6 py-3 rounded-xl border transition-all shrink-0 font-bold text-sm ${activeCategory === cat.id ? 'bg-[#FA5D3E] border-[#FA5D3E] text-white shadow-lg shadow-[#FA5D3E]/20' : 'bg-white border-[#E5E7EB] text-[#4B5563] hover:border-[#FA5D3E]'}`}
                     >
                       <span>{CATEGORY_ICONS[cat.name] || CATEGORY_ICONS['Default']}</span>
                       {cat.name}
                     </button>
                   ))}
                </div>
              </div>

              {/* TABS (Popular / Recent) */}
              <div className="space-y-6">
                <div className="flex gap-8 border-b border-[#E5E7EB]">
                   <button className="pb-4 text-sm font-black border-b-2 border-[#FA5D3E] text-[#FA5D3E]">Popular</button>
                   <button className="pb-4 text-sm font-bold text-[#9CA3AF]">Recent</button>
                </div>

                {/* ITEM GRID */}
                <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-8">
                  {display.data?.map((item: any) => (
                    <motion.div 
                      key={item.id}
                      initial={{ opacity: 0, y: 10 }}
                      animate={{ opacity: 1, y: 0 }}
                      className="bg-white p-5 rounded-[2.5rem] border border-[#E5E7EB] hover:border-[#FA5D3E] transition-all shadow-sm hover:shadow-xl group"
                    >
                      <div className="relative aspect-[4/3] mb-5 overflow-hidden rounded-[2rem]">
                         <img 
                          src={item.image ? (item.image.startsWith('http') ? item.image : `${SERVER_URL}${item.image}`) : 'https://placehold.co/400x300?text=No+Image'} 
                          className="w-full h-full object-cover transition-transform group-hover:scale-110" 
                          alt={item.name} 
                         />
                         <div className="absolute top-4 left-4 w-10 h-10 bg-white/90 backdrop-blur rounded-xl flex items-center justify-center">
                            <input type="checkbox" className="w-4 h-4 accent-[#FA5D3E]" />
                         </div>
                      </div>
                      <div className="space-y-4">
                        <div className="flex items-center justify-between">
                           <h3 className="font-black text-base text-slate-800 tracking-tight">{item.name}</h3>
                           <div className="flex items-center gap-1">
                              <Star size={12} className="text-yellow-400 fill-yellow-400" />
                              <span className="text-[11px] font-bold text-[#9CA3AF]">2.5k+</span>
                           </div>
                        </div>
                        <div className="flex items-center gap-3">
                           <span className="text-xl font-black text-[#FA5D3E]">KES {item.price}</span>
                           <span className="text-xs font-bold text-[#9CA3AF] line-through">KES {Math.round(item.price * 1.2)}</span>
                        </div>
                        <div className="flex gap-3 pt-2">
                           <button className="flex-1 py-3 border border-[#E5E7EB] rounded-xl text-xs font-black uppercase text-[#4B5563] hover:bg-slate-50">Wishlist</button>
                           <button onClick={() => addToCart(item)} className="flex-2 bg-[#FA5D3E] hover:bg-[#E54D32] text-white py-3 rounded-xl text-xs font-black uppercase shadow-lg shadow-[#FA5D3E]/20">Order Now</button>
                        </div>
                      </div>
                    </motion.div>
                  ))}
                </div>
              </div>
            </div>
          )}

          {activeView === 'roster' && (
             <div className="bg-white rounded-3xl p-8 border border-[#E5E7EB] shadow-sm">
               <DutyRoster />
             </div>
          )}

          {activeView !== 'menu' && activeView !== 'roster' && (
            <div className="flex items-center justify-center h-full opacity-20 flex-col">
               <div className="w-20 h-20 bg-slate-200 rounded-2xl mb-4" />
               <p className="font-black uppercase tracking-widest text-xs">{activeView} view coming soon</p>
            </div>
          )}
        </div>
      </main>

      {/* RIGHT SIDEBAR: INVOICE / TRAY */}
      <aside className="w-[380px] bg-white border-l border-[#E5E7EB] flex flex-col shrink-0">
        <div className="p-8 border-b border-[#E5E7EB]">
           <h2 className="text-xl font-black tracking-tight mb-8">Invoice</h2>
           
           <div className="space-y-6 max-h-[40vh] overflow-y-auto pr-2 custom-scrollbar">
              <AnimatePresence>
                {cart.map((item) => (
                  <motion.div 
                    key={item.id}
                    layout
                    initial={{ opacity: 0, x: 20 }}
                    animate={{ opacity: 1, x: 0 }}
                    exit={{ opacity: 0, x: -20 }}
                    className="flex items-center gap-4 group"
                  >
                    <div className="w-16 h-16 bg-[#F9FAFB] rounded-2xl overflow-hidden shrink-0 border border-[#E5E7EB]">
                      <img src={item.image ? (item.image.startsWith('http') ? item.image : `${SERVER_URL}${item.image}`) : 'https://placehold.co/100x100'} className="w-full h-full object-cover" alt="" />
                    </div>
                    <div className="flex-1 min-w-0">
                       <h4 className="text-[13px] font-black text-slate-800 truncate mb-1">{item.name}</h4>
                       <p className="text-[14px] font-black text-[#FA5D3E]">KES {item.price}</p>
                    </div>
                    <div className="flex flex-col items-end gap-2">
                       <button onClick={() => removeFromCart(item.id)} className="text-slate-300 group-hover:text-red-500 transition-colors"><X size={14} /></button>
                       <div className="flex items-center gap-3 bg-[#F9FAFB] border border-[#E5E7EB] rounded-lg px-2 py-1">
                          <button onClick={() => updateQuantity(item.id, -1)} className="text-[#9CA3AF]"><Minus size={12} /></button>
                          <span className="text-xs font-black w-4 text-center">{item.quantity}</span>
                          <button onClick={() => updateQuantity(item.id, 1)} className="text-[#9CA3AF]"><Plus size={12} /></button>
                       </div>
                    </div>
                  </motion.div>
                ))}
              </AnimatePresence>
              {cart.length === 0 && (
                <div className="text-center py-10">
                   <div className="w-16 h-16 bg-slate-50 rounded-2xl flex items-center justify-center mx-auto mb-4 opacity-20">
                      <ShoppingBag size={24} className="text-slate-500" />
                   </div>
                   <p className="text-[11px] font-bold text-[#9CA3AF] uppercase">Empty Tray</p>
                </div>
              )}
           </div>
        </div>

        <div className="p-8 space-y-8 flex-1 flex flex-col justify-end bg-[#F9FAFB]/50">
           <div className="bg-white border border-[#E5E7EB] border-dashed rounded-2xl p-6 space-y-4">
              <div className="flex justify-between items-center text-sm">
                 <span className="font-bold text-[#9CA3AF]">Sub Total</span>
                 <span className="font-black">KES {subTotal.toLocaleString()}</span>
              </div>
              <div className="flex justify-between items-center text-sm">
                 <span className="font-bold text-[#9CA3AF]">Tax</span>
                 <span className="font-black">KES {tax.toLocaleString()}</span>
              </div>
              <div className="h-[1px] bg-[#E5E7EB] border-dashed border-t" />
              <div className="flex justify-between items-center pt-2">
                 <span className="text-base font-black">Total Payment</span>
                 <span className="text-2xl font-black text-[#1F2937]">KES {total.toLocaleString()}</span>
              </div>
           </div>

           <div className="space-y-4">
              <p className="text-[11px] font-black text-[#9CA3AF] uppercase tracking-widest">Payment Method</p>
              <div className="grid grid-cols-2 gap-4">
                 <button className="flex items-center justify-center p-5 bg-white border-2 border-[#E5E7EB] rounded-2xl hover:border-[#FA5D3E] hover:bg-[#FA5D3E]/5 transition-all grayscale hover:grayscale-0">
                    <img src="https://upload.wikimedia.org/wikipedia/commons/thumb/1/15/Mpesa.svg/1200px-Mpesa.svg.png" className="h-6 object-contain" alt="mpesa" />
                 </button>
                 <button className="flex items-center justify-center p-5 bg-white border-2 border-[#FA5D3E] rounded-2xl bg-[#FA5D3E]/5 transition-all">
                    <CardIcon className="text-[#FA5D3E]" size={28} />
                 </button>
                 <button className="flex items-center justify-center p-5 bg-white border-2 border-[#E5E7EB] rounded-2xl hover:border-[#FA5D3E] hover:bg-[#FA5D3E]/5 transition-all text-slate-300 hover:text-[#FA5D3E]">
                    <Banknote size={28} />
                 </button>
                 <button className="flex items-center justify-center p-5 bg-white border-2 border-[#E5E7EB] rounded-2xl hover:border-[#FA5D3E] hover:bg-[#FA5D3E]/5 transition-all text-slate-300 hover:text-[#FA5D3E]">
                    <Wallet size={28} />
                 </button>
              </div>
           </div>

           <button disabled={cart.length === 0} className="w-full py-5 bg-[#FA5D3E] hover:bg-[#E54D32] text-white rounded-2xl text-sm font-black uppercase tracking-widest shadow-xl shadow-[#FA5D3E]/20 transition-all disabled:bg-[#9CA3AF]/20 disabled:shadow-none flex items-center justify-center gap-3">
             Place An Order Now
           </button>
        </div>
      </aside>
    </div>
  );
};

// Simple stub for missing icons in lucide
const ShoppingBag = ({ size, className }: any) => (
  <svg width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className}>
    <path d="M6 2 3 6v14a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2V6l-3-4Z"/><path d="M3 6h18"/><path d="M16 10a4 4 0 0 1-8 0"/>
  </svg>
);

export default WaiterDashboard;
